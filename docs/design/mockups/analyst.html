SELECT TOP 10
  c.Name,
  SUM(o.TotalAmount) AS PurchaseValue
FROM Sales_Fact.[Order] o
JOIN Sales_Dim.Customer c ON o.CustomerID = c.ID
JOIN Sales_Dim.Date d ON o.DateID = d.ID
WHERE d.JalaliYear = 1403
GROUP BY c.Name
ORDER BY PurchaseValue DESC
