# skills/registry.py
class SkillRegistry:
    """Central registry for all available skills."""
    
    def __init__(self):
        self._skills: dict[str, Skill] = {}
    
    def register(self, skill: Skill) -> None:
        if skill.name in self._skills:
            raise ValueError(f"Skill {skill.name} already registered")
        self._skills[skill.name] = skill
    
    def get_applicable_skills(self, context: SkillContext) -> list[Skill]:
        """Return skills that apply to this context, sorted by priority."""
        applicable = [s for s in self._skills.values() if s.applies_to(context)]
        return sorted(applicable, key=lambda s: s.priority)

class FactCheckRefiner(Refiner):
    """Verify output against retrieved facts."""
    
    @property
    def name(self) -> str:
        return "fact_check"
    
    async def refine(self, context: RefinementContext) -> RefinementResult:
        warnings = []
        
        # Extract claims from output
        claims = self._extract_claims(context.raw_output)
        
        # Verify against retrieval context
        for claim in claims:
            if not self._verify_claim(claim, context.retrieval_context):
                warnings.append(f"Unverified claim: {claim}")
        
        # If too many warnings, flag for human review
        confidence = 1.0 - (len(warnings) * 0.2)
        
        return RefinementResult(
            refined_output=context.raw_output,  # No changes, just validation
            confidence=max(0.0, confidence),
            applied_refinements=[self.name],
            warnings=warnings
        )

class FormattingRefiner(Refiner):
    """Improve output formatting and structure."""
    
    @property
    def name(self) -> str:
        return "formatting"
    
    async def refine(self, context: RefinementContext) -> RefinementResult:
        output = context.raw_output
        
        # Add markdown formatting
        output = self._add_headers(output)
        output = self._format_numbers(output)
        output = self._add_bullet_points(output)

# refiner/pipeline.py
class RefinementPipeline:
    """Orchestrates multiple refiners in sequence."""
    
    def __init__(self, refiners: list[Refiner]):
        self.refiners = refiners
    
    async def refine(self, context: RefinementContext) -> RefinementResult:
        current_output = context.raw_output
        all_refinements = []
        all_warnings = []
        min_confidence = 1.0
        
        for refiner in self.refiners:
            if not refiner.enabled:
                continue
            
            # Update context with current output
            current_context = RefinementContext(
                original_question=context.original_question,
                raw_output=current_output,
                retrieval_context=context.retrieval_context,
                metadata=context.metadata
            )
            
            result = await refiner.refine(current_context)
            
            current_output = result.refined_output
            all_refinements.extend(result.applied_refinements)
            all_warnings.extend(result.warnings)
            min_confidence = min(min_confidence, result.confidence)
        
        return RefinementResult(
            refined_output=current_output,
            confidence=min_confidence,
            applied_refinements=all_refinements,
            warnings=all_warnings
        )

"""Base classes for the skills system."""

from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Any

from core.models import RetrievalContext


@dataclass
class SkillContext:
    """Context passed to skills during execution.
    
    Attributes
    ----------
    question : str
        The user's natural-language question.
    retrieval_context : RetrievalContext
        Retrieved entities, facts, and schema information.
    metadata : dict[str, Any]
        Additional metadata that skills can read/write to share information.
    """
    question: str
    retrieval_context: RetrievalContext
    metadata: dict[str, Any]


class Skill(ABC):
    """Abstract base class for all skills.
    
    A skill is a modular capability that can analyze a question and context,
    then optionally modify the context or add metadata for downstream processing.
    
    Skills should be stateless and idempotent where possible.
    """
    
    @property
    @abstractmethod
    def name(self) -> str:
        """Unique identifier for this skill."""
        pass
    
    @property
    @abstractmethod
    def description(self) -> str:
        """Human-readable description of what this skill does."""
        pass
    
    @abstractmethod
    def can_handle(self, context: SkillContext) -> bool:
        """Determine if this skill is applicable to the given context.
        
        Parameters
        ----------
        context : SkillContext
            The current question and retrieval context.
            
        Returns
        -------
        bool
            True if this skill should be executed for this context.
        """
        pass
    
    @abstractmethod
    def execute(self, context: SkillContext) -> SkillContext:
        """Execute the skill's logic and return updated context.
        
        Parameters
        ----------
        context : SkillContext
            The current question and retrieval context.
            
        Returns
        -------
        SkillContext
            Updated context with any modifications or added metadata.
            
        Raises
        ------
        RuntimeError
            If skill execution fails unrecoverably.
        """
        pass


class SkillRegistry:
    """Registry for managing and discovering skills.
    
    The registry maintains a collection of skills and provides methods
    to execute applicable skills for a given context.
    """
    
    def __init__(self):
        """Initialize an empty skill registry."""
        self._skills: list[Skill] = []
    
    def register(self, skill: Skill) -> None:
        """Register a skill with the registry.
        
        Parameters
        ----------
        skill : Skill
            The skill instance to register.
            
        Raises
        ------
        ValueError
            If a skill with the same name is already registered.
        """
        if any(s.name == skill.name for s in self._skills):
            raise ValueError(f"Skill '{skill.name}' is already registered")
        self._skills.append(skill)
    
    def get_applicable_skills(self, context: SkillContext) -> list[Skill]:
        """Get all skills that can handle the given context.
        
        Parameters
        ----------
        context : SkillContext
            The current question and retrieval context.
            
        Returns
        -------
        list[Skill]
            List of skills where can_handle() returns True.
        """
        return [skill for skill in self._skills if skill.can_handle(context)]
    
    def execute_all(self, context: SkillContext) -> SkillContext:
        """Execute all applicable skills in registration order.
        
        Each skill receives the context returned by the previous skill,
        allowing skills to build on each other's work.
        
        Parameters
        ----------
        context : SkillContext
            The initial question and retrieval context.
            
        Returns
        -------
        SkillContext
            Context after all applicable skills have executed.
        """
        applicable = self.get_applicable_skills(context)
        
        for skill in applicable:
            context = skill.execute(context)
        
        return context
    
    def list_skills(self) -> list[tuple[str, str]]:
        """List all registered skills.
        
        Returns
        -------
        list[tuple[str, str]]
            List of (name, description) tuples for all registered skills.
        """
        return [(skill.name, skill.description) for skill in self._skills]

#### # skills/base.py  
#### from abc import ABC, abstractmethod  
#### from dataclasses import dataclass  
####   
#### @dataclass  
#### class SkillContext:  
####     """Shared context passed to all skills."""  
####     question: str  
####     retrieval_context: RetrievalContext  
####     metadata: dict[str, Any]  
####   
#### class Skill(ABC):  
####     """A reusable prompt module that adds specific capability."""  
####   
####     @property  
####     @abstractmethod  
####     def name(self) -> str:  
####         """Unique skill identifier."""  
####         pass  
####   
####     @abstractmethod  
####     def applies_to(self, context: SkillContext) -> bool:  
####         """Return True if this skill should be activated."""  
####         pass  
####   
####     @abstractmethod  
####     def render(self, context: SkillContext) -> str:  
####         """Generate the prompt fragment for this skill."""  
####         pass  
####   
####     @property  
####     def priority(self) -> int:  
####         """Execution order (lower = earlier). Default 100."""  
####         return 100  
####   
####   
####   
#### # skills/registry.py  
#### class SkillRegistry:  
####     """Central registry for all available skills."""  
####   
####     def __init__(self):  
####         self._skills: dict[str, Skill] = {}  
####   
####     def register(self, skill: Skill) -> None:  
####         if skill.name in self._skills:  
####             raise ValueError(f"Skill {skill.name} already registered")  
####         self._skills[skill.name] = skill  
####   
####     def get_applicable_skills(self, context: SkillContext) -> list[Skill]:  
####         """Return skills that apply to this context, sorted by priority."""  
####         applicable = [s for s in self._skills.values() if s.applies_to(context)]  
####         return sorted(applicable, key=lambda s: s.priority)

#### class SkillAwarePromptBuilder:  
####     """Prompt builder that composes skills dynamically."""  
####   
####     def __init__(self, registry: SkillRegistry):  
####         self.registry = registry  
####   
####     def build(  
####         self,  
####         question: str,  
####         base_system_prompt: str,  
####         context: RetrievalContext,  
####     ) -> str:  
####         skill_context = SkillContext(  
####             question=question,  
####             retrieval_context=context,  
####             metadata={}  
####         )  
####   
####         # Get applicable skills  
####         skills = self.registry.get_applicable_skills(skill_context)  
####   
####         # Compose prompt  
####         parts = [base_system_prompt]  
####   
####         for skill in skills:  
####             parts.append(skill.render(skill_context))  
####   
####         parts.append(f"\n## User Question\n{question}")  
####   
####         return "\n\n".join(parts)

#### # refiner/base.py  
#### from abc import ABC, abstractmethod  
#### from dataclasses import dataclass  
####   
#### @dataclass  
#### class RefinementContext:  
####     """Context for refinement operations."""  
####     original_question: str  
####     raw_output: str  
####     retrieval_context: RetrievalContext  
####     metadata: dict[str, Any]  
####   
#### @dataclass  
#### class RefinementResult:  
####     """Result of refinement operation."""  
####     refined_output: str  
####     confidence: float  # 0.0 to 1.0  
####     applied_refinements: list[str]  
####     warnings: list[str]  
####   
#### class Refiner(ABC):  
####     """Base class for output refiners."""  
####   
####     @property  
####     @abstractmethod  
####     def name(self) -> str:  
####         pass  
####   
####     @abstractmethod  
####     async def refine(self, context: RefinementContext) -> RefinementResult:  
####         """Apply refinement to the output."""  
####         pass  
####   
####     @property  
####     def enabled(self) -> bool:  
####         """Whether this refiner is enabled. Override for conditional logic."""  
####         return True  
####   
####   
####   
#### # refiner/implementations.py  
#### class FactCheckRefiner(Refiner):  
####     """Verify output against retrieved facts."""  
####   
####     @property  
####     def name(self) -> str:  
####         return "fact_check"  
####   
####     async def refine(self, context: RefinementContext) -> RefinementResult:  
####         warnings = []  
####   
####         # Extract claims from output  
####         claims = self._extract_claims(context.raw_output)  
####   
####         # Verify against retrieval context  
####         for claim in claims:  
####             if not self._verify_claim(claim, context.retrieval_context):  
####                 warnings.append(f"Unverified claim: {claim}")  
####   
####         # If too many warnings, flag for human review  
####         confidence = 1.0 - (len(warnings) * 0.2)  
####   
####         return RefinementResult(  
####             refined_output=context.raw_output,  # No changes, just validation  
####             confidence=max(0.0, confidence),  
####             applied_refinements=[self.name],  
####             warnings=warnings  
####         )  
####   
#### class FormattingRefiner(Refiner):  
####     """Improve output formatting and structure."""  
####   
####     @property  
####     def name(self) -> str:  
####         return "formatting"  
####   
####     async def refine(self, context: RefinementContext) -> RefinementResult:  
####         output = context.raw_output  
####   
####         # Add markdown formatting  
####         output = self._add_headers(output)  
####         output = self._format_numbers(output)  
####         output = self._add_bullet_points(output)  
####   
####         return RefinementResult(  
####             refined_output=output,  
####             confidence=1.0,  
####             applied_refinements=[self.name],  
####             warnings=[]  
####         )  
####   
#### class CitationRefiner(Refiner):  
####     """Add citations to sources."""  
####   
####     @property  
####     def name(self) -> str:  
####         return "citation"  
####   
####     async def refine(self, context: RefinementContext) -> RefinementResult:  
####         output = context.raw_output  
####   
####         # Add footnotes for each table/fact used  
####         citations = self._generate_citations(context.retrieval_context)  
####   
####         if citations:  
####             output += "\n\n## Sources\n" + "\n".join(citations)  
####   
####         return RefinementResult(  
####             refined_output=output,  
####             confidence=1.0,  
####             applied_refinements=[self.name],  
####             warnings=[]  
####         )  
####   
#### class LLMRefiner(Refiner):  
####     """Use a second LLM call to improve output."""  
####   
####     def __init__(self, llm_backend: LLMBackend):  
####         self.llm = llm_backend  
####   
####     @property  
####     def name(self) -> str:  
####         return "llm_refinement"  
####   
####     async def refine(self, context: RefinementContext) -> RefinementResult:  
####         refinement_prompt = f"""  
#### You are an editor. Improve this answer:  
####   
#### Original Question: {context.original_question}  
####   
#### Draft Answer:  
#### {context.raw_output}  
####   
#### Make it:  
#### - More concise  
#### - Better structured  
#### - Easier to understand  
#### - Factually accurate  
####   
#### Improved Answer:"""  
####   
####         try:  
####             refined = await self.llm.generate(refinement_prompt)  
####             return RefinementResult(  
####                 refined_output=refined,  
####                 confidence=0.9,  
####                 applied_refinements=[self.name],  
####                 warnings=[]  
####             )  
####         except Exception as e:  
####             return RefinementResult(  
####                 refined_output=context.raw_output,  
####                 confidence=0.5,  
####                 applied_refinements=[],  
####                 warnings=[f"LLM refinement failed: {e}"]  
####             )  
####   
####   
####   
#### # refiner/pipeline.py  
#### class RefinementPipeline:  
####     """Orchestrates multiple refiners in sequence."""  
####   
####     def __init__(self, refiners: list[Refiner]):  
####         self.refiners = refiners  
####   
####     async def refine(self, context: RefinementContext) -> RefinementResult:  
####         current_output = context.raw_output  
####         all_refinements = []  
####         all_warnings = []  
####         min_confidence = 1.0  
####   
####         for refiner in self.refiners:  
####             if not refiner.enabled:  
####                 continue  
####   
####             # Update context with current output  
####             current_context = RefinementContext(  
####                 original_question=context.original_question,  
####                 raw_output=current_output,  
####                 retrieval_context=context.retrieval_context,  
####                 metadata=context.metadata  
####             )  
####   
####             result = await refiner.refine(current_context)  
####   
####             current_output = result.refined_output  
####             all_refinements.extend(result.applied_refinements)  
####             all_warnings.extend(result.warnings)  
####             min_confidence = min(min_confidence, result.confidence)  
####   
####         return RefinementResult(  
####             refined_output=current_output,  
####             confidence=min_confidence,  
####             applied_refinements=all_refinements,  
####             warnings=all_warnings  
####         )

## Critical Review (Based on Available Summaries)

### Architectural Flaws

1. **No clear layering contract.** You have `api/`, `llm/`, `prompt_engine/`, `retrieval/`, `schema_data/`, `knowledge/` — but the dependency direction between these layers is not enforced anywhere. Nothing stops `retrieval/` from importing `api/` or `prompt_engine/` from importing `llm/`. This will rot over time.

2. **`PromptBuilder` is a static-method class masquerading as a module.** A class with only `@staticmethod` methods and no instance state is just a module with extra ceremony. It adds no value over plain functions in a `builder.py` module.

3. **`ValueRetriever` has the same problem.** All `@staticmethod` — it's not a class, it's a namespace. No polymorphism, no injection, no testability benefit.

4. **`QueryCache` is in-memory only.** A single-process LRU+TTL cache dies on every restart and doesn't scale horizontally. There's no abstraction layer (interface) over it, so swapping to Redis later requires touching call sites.

5. **`get_settings()` uses `@lru_cache`.** This is a well-known anti-pattern for settings in production — it makes testing painful (you can't swap settings between tests without cache-busting hacks) and hides the dependency.

6. **`LLMBackend` raises `ValueError("OUT_OF_SCOPE")` as a control-flow sentinel.** Using a stringly-typed exception for business logic is fragile. Any typo in the sentinel string silently breaks the contract. This should be a dedicated exception class.

7. **No pipeline/orchestration abstraction.** The flow `question → retrieval → prompt → LLM → response` is presumably wired together somewhere, but there's no visible `Pipeline` or `Agent` class that owns this sequence. This makes it impossible to insert steps (like a refiner) cleanly.

8. **`_load_validated` in `config_loader.py` is a private function doing public work.** Config loading is a core concern; hiding it behind a `_` prefix signals it was never meant to be a stable API, yet it presumably drives the whole knowledge layer.

### Code Quality Issues

1. **`_IdfDict` subclasses `dict`.** Subclassing built-in `dict` is almost always wrong — you get surprising behavior with methods like `update`, `copy`, `|=`. Use `collections.UserDict` or composition.

2. **`_forced_tables` and `_score_table` are module-level private functions in `schema_data/retriever.py`.** If this module grows, these become impossible to test in isolation or override. They should be methods on a `SchemaRetriever` class.

3. **`_normalise` and `_tokenize` are also module-level privates.** These are pure utility functions that likely belong in a shared `utils/text.py` — they're probably duplicated elsewhere.

4. **`RING_ALIASES` is a hardcoded constant** (implied by `ValueRetriever`). Domain aliases belong in config/YAML, not source code. Every alias change requires a code deploy.

5. **`RetrievalContext` ordering caveat is documented but not enforced.** The `PromptBuilder` docstring explicitly warns that `set()` destroys insertion order, then says "use `context.selected_tables` instead." This is a latent bug waiting to happen — the code should not offer two inconsistent paths.

6. **`QueryRequest` validates `min_length=1` AND has a `question_not_blank` validator.** These are redundant. Pick one.

### Security Vulnerabilities

1. **No visible input sanitization before SQL generation.** If the question flows into a SQL prompt, prompt injection is a real attack surface. There's no mention of any sanitization layer between `QueryRequest` and `PromptBuilder`.

2. **`CacheInvalidateRequest` exists** — who can call it? If this endpoint is unauthenticated, any caller can flush your cache. No auth layer is visible in the summaries.

3. **`load_yaml` in `config_loader.py`** — if this uses `yaml.load()` instead of `yaml.safe_load()`, it's a remote code execution vulnerability.

4. **LLM output is described as "raw text output (no cleaning)"** in `LLMBackend`. If this raw output is ever rendered in a UI or interpolated into further prompts without sanitization, you have both XSS and prompt injection risks.

## Revised Critical Findings (Now With Full Context)

### Newly Visible Problems

1. **`llm/ollama_client.py` and `llm/sql_agent.py` both exist and both do SQL generation.** `ollama_client.py` is described as a "backward-compatible shim" but it constructs its own `OllamaBackend` instance at module level (`_backend = OllamaBackend()`). 2. **`SQLAgent._generate_and_clean` returns `(sql, raw, [])` — always an empty list.** The third return value is hardcoded to `[]` every call. The correction prompt accumulation in `run()` appends to a separate local list. This is misleading and the return signature is a lie.

3. **`PromptBuilder.build` uses `set()` to deduplicate tables**, destroying insertion order. `context.selected_tables` (on `RetrievalContext`) does order-preserving dedup correctly. `PromptBuilder` ignores it and does the wrong thing independently. This is a direct contradiction between two parts of the same codebase.

4. **`PROMPT_TEMPLATE` has no skill or refiner slots.** Every section is hardcoded. Adding skills currently requires either editing the template string directly or bolting text onto `system_prompt` — both are hacks.

5. **`RetrievalContext` has both `entities` and `dimensions` fields that are set to the same value** in `ContextRetriever.retrieve()`: `dimensions=entities`. This is pure redundancy with no enforcement that they stay in sync. One of them will drift.

6. **`WizardLLM` and `OllamaBackend` are parallel implementations** of the same concept with no shared interface between them. `WizardLLM` supports four providers; `OllamaBackend` supports one. `SQLAgent` uses `OllamaBackend`. `WizardLLM` appears to be used elsewhere. These two hierarchies will permanently diverge.

7. **`override_settings` in `config.py` manually iterates `__slots__`** to copy fields. If a new field is added to `Settings` without updating `override_settings`, the context manager silently drops it. This is a maintenance trap.

8. **`QueryResponse` uses `list[dict[str, Any]]` for `result`.** There is no schema enforcement on what those dicts contain. A refiner or downstream consumer has no type safety on result rows.

**`skills/registry.py`:**
```python
from __future__ import annotations
from core.models import RetrievalContext
from skills.base import Skill

_REGISTRY: dict[str, type[Skill]] = {}


def register(skill_cls: type[Skill]) -> type[Skill]:
    """Decorator — registers a Skill subclass by its ``name`` attribute."""
    if not hasattr(skill_cls, "name"):
        raise TypeError(f"{skill_cls.__name__} must define a class-level 'name'")
    _REGISTRY[skill_cls.name] = skill_cls
    return skill_cls


def select_skills(
    question: str,
    context: RetrievalContext,
    names: list[str] | None = None,
) -> list[Skill]:
    """Return instantiated skills applicable to this question/context.

    Parameters
    ----------
    names:
        If supplied, only skills whose ``name`` is in this list are
        considered. Otherwise all registered skills are evaluated.
    """
    pool = (
        {n: _REGISTRY[n] for n in names if n in _REGISTRY}
        if names
        else _REGISTRY
    )
    return [
        cls() for cls in pool.values()
        if cls().is_applicable(question, context)
    ]
```

class Refiner(ABC):

    Refiners are stateless. All context needed must come from the
    arguments to ``refine()``.
    """

    @abstractmethod
    def refine(
        self,
        question: str,
        raw_llm_output: str,
        response: QueryResponse,

        Must never raise — log and return the input unchanged on error.

**`refiners/chain.py`:**
```python
from __future__ import annotations
import logging
from api.models import QueryResponse
from refiners.base import Refiner

logger = logging.getLogger(__name__)


class RefinerChain:
    """Runs a sequence of Refiners, passing output of each as input to next."""

    def __init__(self, refiners: list[Refiner]) -> None:
        self._refiners = refiners

    def run(
        self,
        question: str,
        raw_llm_output: str,
        response: QueryResponse,
    ) -> QueryResponse:
        for refiner in self._refiners:
            try:
                response = refiner.refine(question, raw_llm_output, response)
            except Exception:
                logger.exception(
                    "Refiner %s failed — skipping",
                    type(refiner).__name__,
                )
        return response

**`refiners/sql_refiner.py`** — this is where the key logic lives:
```python
from __future__ import annotations
import logging
import re
from api.models import QueryResponse
from refiners.base import Refiner

logger = logging.getLogger(__name__)

# Dangerous patterns that should never appear in generated SQL
_FORBIDDEN = re.compile(
    r"\b(INSERT|UPDATE|DELETE|DROP|TRUNCATE|ALTER|EXEC|EXECUTE|xp_)\b",
    re.IGNORECASE,
)


class SQLSanitizingRefiner(Refiner):
    """Strips dangerous statements from generated SQL.
    Complements security/sql_guard.py — does not replace it."""

    def refine(self, question, raw_llm_output, response):
        if not response.sql:
            return response
        if _FORBIDDEN.search(response.sql):
            logger.warning(
                "SQLSanitizingRefiner: forbidden pattern in SQL — nulling out. SQL: %.200s",
                response.sql,
            )
            return response.model_copy(update={"sql": None})
        return response


class SQLFormattingRefiner(Refiner):
    """Normalizes whitespace and removes markdown artifacts from SQL."""

    def refine(self, question, raw_llm_output, response):
        if not response.sql:
            return response
        sql = response.sql.strip()
        # Remove any residual markdown fences the LLM snuck in
        sql = re.sub(r"^```(?:sql)?\s*", "", sql, flags=re.IGNORECASE)
        sql = re.sub(r"\s*```$", "", sql)
        sql = sql.strip()
        return response.model_copy(update={"sql": sql})
```

import logging
from api.models import QueryResponse
from refiners.base import Refiner

logger = logging.getLogger(__name__)


class RowCapRefiner(Refiner):
    """Enforces max_rows_returned cap on result sets."""

    def __init__(self, max_rows: int) -> None:
        self._max_rows = max_rows

    def refine(self, question, raw_llm_output, response):
        if response.result and len(response.result) > self._max_rows:
            logger.warning(
                "RowCapRefiner: truncating %d rows to %d",
                len(response.result),
                self._max_rows,
            )
            truncated = response.result[: self._max_rows]
            return response.model_copy(update={
                "result": truncated,
                "row_count": len(truncated),
            })
        return response

### `skills/base.py`

```python
"""Abstract base for all Skills.

A Skill is a reusable prompt module that injects capability-specific
instructions and context into the prompt pipeline.

To add a new skill:
1. Subclass Skill and set a unique class-level ``name``.
2. Implement ``system_fragment`` and ``context_fragment``.
3. Decorate with ``@register`` from ``skills.registry``.
"""
from __future__ import annotations

from abc import ABC, abstractmethod

from core.models import RetrievalContext


class Skill(ABC):
    """Contract every Skill must satisfy."""

    name: str  # must be set on every concrete subclass

    @abstractmethod
    def system_fragment(self) -> str:
        """Return a system-prompt fragment this skill contributes.

        Return empty string if this skill adds nothing to the system block.
        """

    @abstractmethod
    def context_fragment(self, context: RetrievalContext) -> str:
        """Return a context block injected into the prompt body.

        Return empty string if this skill adds nothing to the context block.
        """

    def is_applicable(self, question: str, context: RetrievalContext) -> bool:
        """Gate skill activation.

        Default: always active. Override to restrict to specific question
        types or context shapes.
        """
        return True
```

### `skills/registry.py`

```python
"""Skill registration and selection.

Usage::

    from skills.registry import register, select_skills

    @register
    class MySkill(Skill):
        name = "my_skill"
        ...

    active = select_skills(question, context)
"""
from __future__ import annotations

import logging

from core.models import RetrievalContext
from skills.base import Skill

logger = logging.getLogger(__name__)

_REGISTRY: dict[str, type[Skill]] = {}


def register(skill_cls: type[Skill]) -> type[Skill]:
    """Decorator — registers a Skill subclass by its ``name`` attribute.

    Raises
    ------
    TypeError
        If the class does not define a ``name`` attribute.
    """
    if not hasattr(skill_cls, "name"):
        raise TypeError(
            f"{skill_cls.__name__} must define a class-level 'name' string."
        )
    if skill_cls.name in _REGISTRY:
        logger.warning(
            "Skill %r is already registered — overwriting with %s.",
            skill_cls.name,
            skill_cls.__name__,
        )
    _REGISTRY[skill_cls.name] = skill_cls
    return skill_cls


def select_skills(
    question: str,
    context: RetrievalContext,
    names: list[str] | None = None,
) -> list[Skill]:
    """Return instantiated skills applicable to this question and context.

    Parameters
    ----------
    question:
        The natural-language question.
    context:
        The fully populated RetrievalContext for this question.
    names:
        If supplied, only skills whose ``name`` is in this list are
        considered. Otherwise all registered skills are evaluated.
    """
    pool = (
        {n: _REGISTRY[n] for n in names if n in _REGISTRY}
        if names is not None
        else _REGISTRY
    )
    active: list[Skill] = []
    for cls in pool.values():
        instance = cls()
        try:
            if instance.is_applicable(question, context):
                active.append(instance)
        except Exception:  # noqa: BLE001
            logger.exception(
                "Skill %r raised in is_applicable — skipping.", cls.name
            )
    return active
```

* Refiners are stateless — all context comes from method arguments.

    """Contract every Refiner must satisfy."""

        Parameters
        ----------
        question:
            The original natural-language question.
        raw_llm_output:
            Verbatim text returned by the LLM before any cleaning.
            Useful for refiners that need to inspect the raw response.
        response:
            The QueryResponse built from the cleaned, executed result.

        Returns
        -------
        QueryResponse
            Either the input unchanged or a new instance via model_copy.
        """

```python
"""Sequential refiner runner."""
from __future__ import annotations

import logging

from api.models import QueryResponse
from refiners.base import Refiner

logger = logging.getLogger(__name__)


class RefinerChain:
    """Runs a list of Refiners in order, passing each output as the next input.

    A failing refiner is logged and skipped — it never aborts the chain.
    """

    def __init__(self, refiners: list[Refiner]) -> None:
        self._refiners = refiners

    def run(
        self,
        question: str,
        raw_llm_output: str,
        response: QueryResponse,
    ) -> QueryResponse:
        for refiner in self._refiners:
            try:
                response = refiner.refine(question, raw_llm_output, response)
            except Exception:
                logger.exception(
                    "Refiner %s raised unexpectedly — skipping.",
                    type(refiner).__name__,
                )
        return response
```

### `refiners/sql_refiner.py`

This deliberately delegates to `sql_guard` rather than re-implementing checks.

```python
"""SQL-focused refiners.

These refiners operate on QueryResponse.sql. They delegate to
security.sql_guard for all validation and cleaning logic — they do
not re-implement those checks.
"""
from __future__ import annotations

import logging

from api.models import QueryResponse
from refiners.base import Refiner
from security.sql_guard import clean_sql, validate_sql, ensure_top

logger = logging.getLogger(__name__)


class SQLCleaningRefiner(Refiner):
    """Re-runs clean_sql on the SQL field as a defence-in-depth measure.

    clean_sql is already called in SQLAgent._generate_and_clean, so this
    refiner is a safety net for any path that bypasses the agent.
    """

    def refine(self, question, raw_llm_output, response):
        if not response.sql:
            return response
        try:
            cleaned = clean_sql(response.sql)
            return response.model_copy(update={"sql": cleaned})
        except ValueError:
            logger.warning(
                "SQLCleaningRefiner: clean_sql failed — nulling SQL. "
                "Raw: %.200s",
                response.sql,
            )
            return response.model_copy(update={"sql": None})


class SQLValidationRefiner(Refiner):
    """Validates SQL against security rules. Nulls out SQL on violation.

    Delegates entirely to security.sql_guard.validate_sql.
    """

    def refine(self, question, raw_llm_output, response):
        if not response.sql:
            return response
        try:
            validate_sql(response.sql)
            return response
        except ValueError as exc:
            logger.warning(
                "SQLValidationRefiner: validation failed (%s) — nulling SQL.",
                exc,
            )
            return response.model_copy(update={"sql": None})


class TopNRefiner(Refiner):
    """Injects TOP n into SQL when no row-limit clause is present.

    Fixes the live bug where ensure_top is defined in sql_guard but
    never called in the main execution path.
    """

    def __init__(self, max_rows: int) -> None:
        self._max_rows = max_rows

    def refine(self, question, raw_llm_output, response):
        if not response.sql:
            return response
        capped = ensure_top(response.sql, self._max_rows)
        return response.model_copy(update={"sql": capped})
```

from __future__ import annotations

import logging

from api.models import QueryResponse
from refiners.base import Refiner

logger = logging.getLogger(__name__)


class RowCapRefiner(Refiner):
    """Truncates result rows to max_rows as a last-resort safety net."""

    def __init__(self, max_rows: int) -> None:
        self._max_rows = max_rows

    def refine(self, question, raw_llm_output, response):
        if not response.result:
            return response
        if len(response.result) > self._max_rows:
            logger.warning(
                "RowCapRefiner: truncating %d rows to %d.",
                len(response.result),
                self._max_rows,
            )
            truncated = response.result[: self._max_rows]
            return response.model_copy(update={
                "result": truncated,
                "row_count": len(truncated),
            })
        return response

    def refine(self, question, raw_llm_output, response):
        if response.result is not None and len(response.result) == 0:
            return response.model_copy(update={"result": None, "row_count": 0})
        return response
